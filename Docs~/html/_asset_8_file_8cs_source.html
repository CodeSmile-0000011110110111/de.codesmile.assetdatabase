1109 internal static Boolean CopyInternal([NotNull] Path sourcePath, [NotNull] Path destinationPath,
1110 Boolean overwriteExisting)
1111 {
1112 ThrowIf.ArgumentIsNull(sourcePath, nameof(sourcePath));
1113 ThrowIf.ArgumentIsNull(destinationPath, nameof(destinationPath));
1114 ThrowIf.AssetPathNotInDatabase(sourcePath);
1115 ThrowIf.SourceAndDestPathAreEqual(sourcePath, destinationPath);
1116
1117 destinationPath.CreateFolders();
1118
1119 var success = AssetDatabase.CopyAsset(sourcePath, destinationPath);
1120 SetLastErrorMessage(success ? String.Empty : $"failed to copy {sourcePath} to {destinationPath}");
1121 return success;
1122 }
